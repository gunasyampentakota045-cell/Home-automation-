function showPage(page) {
  // hide all pages
  document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
  // show/hide nav
  const isAuth = page !== 'login';
  document.getElementById('topnav').classList.toggle('hidden', !isAuth);
  document.getElementById('bottomnav').classList.toggle('hidden', !isAuth);
  // show target page
  document.getElementById('page-'+page).classList.add('active');
  if (page !== 'login') prevPage = currentPage;
  currentPage = page;
  // update bottom nav
  document.querySelectorAll('.bottom-nav-item').forEach(b => b.classList.remove('active'));
  const activeBtn = document.getElementById('bn-'+page);
  if (activeBtn) activeBtn.classList.add('active');
  // page-specific init
  if (page === 'home') { startHeroSlider(); }
  else { stopHeroSlider(); }
  if (page === 'groceries') renderCategoryPage('groceries');
  if (page === 'accessories') renderCategoryPage('accessories');
  if (page === 'beauty') renderCategoryPage('beauty');
  if (page === 'cart') renderCart();
  if (page === 'pay') { updateWalletDisplay(); renderTransactions(); }
  if (page === 'offers') renderOffers();
  if (page === 'settings') renderSettings();
  window.scrollTo(0,0);
}

function showLoginStep(step) {
  ['login','otp','forgot'].forEach(s => {
    document.getElementById('step-'+s).classList.toggle('hidden', s !== step);
  });
  if (step === 'otp') startResendTimer();
}

function handleSignIn() {
  const email = document.getElementById('loginEmail').value.trim();
  const pass = document.getElementById('loginPass').value.trim();
  if (!email) { showToast('Please enter your email or phone'); return; }
  if (!pass) { showToast('Please enter your password'); return; }
  userEmail = email;
  updateUserProfile();
  showLoginStep('otp');
}

function handleSignUp() {
  const email = document.getElementById('loginEmail').value.trim();
  if (!email) { showToast('Please enter your email or phone to create an account'); return; }
  userEmail = email;
  updateUserProfile();
  showLoginStep('otp');
}

function updateUserProfile() {
  const avatar = userEmail.charAt(0).toUpperCase();
  const el = document.getElementById('userAvatar');
  const ne = document.getElementById('userName');
  const ee = document.getElementById('userEmail');
  if (el) el.textContent = avatar;
  if (ne) ne.textContent = userEmail.includes('@') ? userEmail.split('@')[0] : userEmail;
  if (ee) ee.textContent = userEmail;
}

function verifyOtp() {
  const inputs = document.querySelectorAll('.otp-input');
  const otp = Array.from(inputs).map(i => i.value).join('');
  if (otp.length < 6) { showToast('Please enter all 6 digits'); return; }
  showToast('Welcome to Real Choice!');
  setTimeout(() => showPage('home'), 600);
}

function otpMove(input, idx) {
  if (input.value.length === 1 && idx < 6) {
    document.querySelectorAll('.otp-input')[idx].focus();
  }
  if (input.value.length === 0 && idx > 1) {
    document.querySelectorAll('.otp-input')[idx-2].focus();
  }
}

function startResendTimer() {
  let t = 30;
  document.getElementById('resendBtn').disabled = true;
  document.getElementById('resendTimer').textContent = t;
  clearInterval(resendInterval);
  resendInterval = setInterval(() => {
    t--;
    document.getElementById('resendTimer').textContent = t;
    if (t <= 0) {
      clearInterval(resendInterval);
      document.getElementById('resendBtn').disabled = false;
    }
  }, 1000);
}

function resendOtp() { showToast('OTP resent!'); startResendTimer(); }

}

  activeCategory[type] = cat;

  document.getElementById(searchId).value = '';

function renderProductGrid(type, searchVal) {
  const gridEl = document.getElementById(type==='groceries'?'groceryGrid':type==='accessories'?'accessoryGrid':'beautyGrid');
  let products = PRODUCTS[type][activeCategory[type]] || [];
  if (searchVal) {
    const q = searchVal.toLowerCase();
    products = products.filter(p => p.name.toLowerCase().includes(q) || p.meta.toLowerCase().includes(q));
  }
  if (products.length === 0) {
    gridEl.innerHTML = `<div style="grid-column:1/-1;text-align:center;padding:3rem;color:hsl(var(--muted-fg))">No products found</div>`;
    return;
  }
  gridEl.innerHTML = products.map(p => productCardHTML(p)).join('');
}

function findProduct(id) {
  for (const section of Object.values(PRODUCTS)) {
    for (const items of Object.values(section)) {
      const found = items.find(p => p.id === id);
      if (found) return found;
    }
  }
  return null;
}

function addToCart(id) {
  const product = findProduct(id);
  if (!product) return;
  const existing = cart.find(c => c.id === id);
  if (existing) { existing.qty++; }
  else { cart.push({...product, qty:1}); }
  saveCart();
  showToast(product.name + ' added to cart');
  refreshProductCards();
}

function updateQty(id, delta) {
  const item = cart.find(c => c.id === id);
  if (!item) return;
  item.qty += delta;
  if (item.qty <= 0) cart = cart.filter(c => c.id !== id);
  saveCart();
  refreshProductCards();
  if (currentPage === 'cart') renderCart();
}

function saveCart() {
  localStorage.setItem('rc_cart', JSON.stringify(cart));
  updateCartBadge();
}

function updateCartBadge() {
  const total = cart.reduce((s,i) => s+i.qty, 0);
  const badge = document.getElementById('cartBadge');
  if (badge) badge.textContent = total;
  const countEl = document.getElementById('cartItemCount');
  if (countEl) countEl.textContent = total + ' item' + (total !== 1 ? 's' : '');
}

function loadSettingsUI() {

  if (dt) dt.classList.toggle('on', settings.dark);
  const nt = document.getElementById('notifToggle');
  if (nt) nt.classList.toggle('on', settings.notif);

  const lang = document.getElementById('langSelect');
  if (lang) lang.value = settings.lang || 'en';
  const cur = document.getElementById('currencySelect');
  if (cur) cur.value = settings.currency || 'INR';
}

  settings.dark = !settings.dark;
  applySettings();
  saveSettings(true);

function updateWalletDisplay() {
  const el = document.getElementById('walletBalance');
  const sym = getCurrencySymbol();
  if (el) el.innerHTML = sym + walletBalance.toFixed(2);
}

function selectAmount(n) {
  selectedAmount = n;
  document.querySelectorAll('.amount-chip').forEach(c => c.classList.remove('selected'));
  event.target.classList.add('selected');
  document.getElementById('customAmount').value = '';
}

function addToWallet() {
  const custom = parseFloat(document.getElementById('customAmount').value);
  const amount = custom > 0 ? custom : selectedAmount;
  if (!amount || amount <= 0) { showToast('Please select or enter an amount'); return; }
  walletBalance += amount;
  localStorage.setItem('rc_wallet', walletBalance.toString());
  updateWalletDisplay();
  TRANSACTIONS.unshift({id:'t'+Date.now(),desc:'Added to Wallet',amount,type:'credit',date:'Just now'});
  renderTransactions();
  document.getElementById('customAmount').value = '';
  document.querySelectorAll('.amount-chip').forEach(c => c.classList.remove('selected'));
  selectedAmount = 0;
  showToast('&#8377;' + amount + ' added to wallet!');
}

function saveCard() {
  const n = document.getElementById('cardNumber').value.replace(/\s/g,'');
  const e = document.getElementById('cardExpiry').value;
  const c = document.getElementById('cardCvv').value;
  const nm = document.getElementById('cardName').value;
  if (n.length < 16) { showToast('Enter a valid 16-digit card number'); return; }
  if (e.length < 5) { showToast('Enter a valid expiry date'); return; }
  if (c.length < 3) { showToast('Enter a valid CVV'); return; }
  if (!nm) { showToast('Enter cardholder name'); return; }
  showToast('Card saved securely!');
  document.getElementById('cardNumber').value = '';
  document.getElementById('cardExpiry').value = '';
  document.getElementById('cardCvv').value = '';
  document.getElementById('cardName').value = '';
}

function formatCard(input) {
  let v = input.value.replace(/\D/g,'').substring(0,16);
  input.value = v.replace(/(.{4})/g,'$1 ').trim();
}

function formatExpiry(input) {
  let v = input.value.replace(/\D/g,'').substring(0,4);
  if (v.length >= 2) v = v.substring(0,2) + '/' + v.substring(2);
  input.value = v;
}

function getCurrencySymbol() {
  const c = settings.currency || 'INR';
  return c === 'USD' ? '$' : c === 'EUR' ? '€' : '₹';
}

function convertPrice(inr) {
  const c = settings.currency || 'INR';
  if (c === 'USD') return (inr / 83).toFixed(2);
  if (c === 'EUR') return (inr / 91).toFixed(2);
  return inr;
}

// ============================================================
function handleGlobalSearch(q) {
  if (!q) return;
  // Find the first matching product across all categories
  for (const [type, cats] of Object.entries(PRODUCTS)) {
    for (const [cat, items] of Object.entries(cats)) {
      if (items.some(p => p.name.toLowerCase().includes(q.toLowerCase()))) {
        activeCategory[type] = cat;
        showPage(type);
        const searchId = type==='groceries'?'grocerySearch':type==='accessories'?'accessorySearch':'beautySearch';
        document.getElementById(searchId).value = q;
        filterProducts(type, q);
        document.getElementById('globalSearch').value = q;
        return;
      }
    }
  }
}

let toastTimer = null;
function showToast(msg) {
  const t = document.getElementById('toast');
  t.innerHTML = `<svg width="16" height="16" viewBox="0 0 24 24" fill="none" stroke="white" stroke-width="2.5"><polyline points="20 6 9 17 4 12"/></svg>${msg}`;
  t.classList.add('show');
  if (toastTimer) clearTimeout(toastTimer);
  toastTimer = setTimeout(() => t.classList.remove('show'), 2800);
}
